def w3_require(value, message):
    if not value:
        raise ValueError(message)

def w3_sha(path):
    digest = hashlib.sha256()
    with pathlib.Path(path).open("rb") as stream:
        for block in iter(lambda: stream.read(1024 * 1024), b""):
            digest.update(block)
    return digest.hexdigest()

def w3_write(path, value):
    path = pathlib.Path(path)
    with path.open("x") as stream:
        stream.write(json.dumps(value, indent=2, sort_keys=True) + "\n")

def w3_file(path):
    path = pathlib.Path(path)
    w3_require(path.is_file() and not path.is_symlink(), "Expected an original regular file: " + str(path))
    return {"bytes": path.stat().st_size, "sha256": w3_sha(path)}

def w3_project_inventory(project):
    project = pathlib.Path(project)
    w3_require(project.is_dir() and not project.is_symlink(), "Generated project is not a physical directory")
    entries = {}
    for path in sorted(project.rglob("*")):
        w3_require(not path.is_symlink(), "Unexpected generated project symlink")
        mode = stat.S_IMODE(path.stat().st_mode)
        relative = str(path.relative_to(project))
        if path.is_file():
            entries[relative] = dict(kind="file", mode=mode, **w3_file(path))
        elif path.is_dir():
            entries[relative] = {"kind": "directory", "mode": mode}
        else:
            raise ValueError("Unexpected generated project node: " + relative)
    w3_require("project.pbxproj" in entries and
               "xcshareddata/xcschemes/PinkhaPrebuiltUI.xcscheme" in entries,
               "Generated UI project and exact scheme are required")
    return {"schema": "pinkha-generated-ui-project-inventory-v1", "root": str(project), "entries": entries}

def w3_sources(relative_paths, source_root, compiled_root, source_before):
    source_root, compiled_root = pathlib.Path(source_root), pathlib.Path(compiled_root)
    w3_require(relative_paths and len(relative_paths) == len(set(relative_paths)), "Selected source paths must be unique")
    rows = []
    for relative in relative_paths:
        path = pathlib.PurePosixPath(relative)
        w3_require(not path.is_absolute() and ".." not in path.parts and path.suffix == ".swift", "Invalid selected Swift source")
        expected = source_before["files"][relative]
        original, compiled = source_root / relative, compiled_root / relative
        w3_require(w3_file(original) == expected == w3_file(compiled), "Compiled source differs: " + relative)
        rows.append(dict(repositoryPath=relative, compiledPath=str(compiled), **expected))
    return rows

def w3_compiler_invocations(log, module, filelist):
    witnesses = []
    with pathlib.Path(log).open(errors="strict") as stream:
        for number, line in enumerate(stream, 1):
            if ".SwiftFileList" not in line or "swiftc" not in line:
                continue
            try:
                tokens = shlex.split(line)
            except ValueError:
                continue
            if not any(pathlib.Path(token).name == "swiftc" for token in tokens):
                continue
            names = [tokens[i + 1] for i, token in enumerate(tokens[:-1]) if token == "-module-name"]
            lists = [token[1:] for token in tokens if token.startswith("@")]
            if names == [module] and any(item.casefold() == str(filelist).casefold() for item in lists):
                witnesses.append({"lineNumber": number,
                                  "lineSHA256": hashlib.sha256(line.rstrip("\r\n").encode()).hexdigest(),
                                  "moduleName": module})

    return witnesses

def w3_compiler_lists(derived, module, required_sources, log, evidence, phase):
    derived, evidence = pathlib.Path(derived), pathlib.Path(evidence)
    expected_paths = [row["compiledPath"] for row in required_sources]
    folded = [path.casefold() for path in expected_paths]
    w3_require(len(folded) == len(set(folded)), "Case-colliding source paths cannot qualify")
    inventory = []
    for path in sorted((derived / "Build/Intermediates.noindex").rglob("*SwiftFileList")):
        if not any(part.casefold() == (module + ".build").casefold() for part in path.parts):
            continue
        if not path.is_file() or path.is_symlink():
            continue
        lines = [line.strip().strip('"') for line in path.read_text().splitlines()]
        normalized = [line.casefold() for line in lines]
        if not all(normalized.count(item) == 1 for item in folded):
            continue
        # Case normalization is permitted only for the same actual APFS source.
        for expected, source_row in zip(expected_paths, required_sources):
            actual = pathlib.Path(lines[normalized.index(expected.casefold())])
            w3_require(actual.is_file() and actual.samefile(expected) and
                       w3_file(actual) == {key: source_row[key] for key in ("bytes", "sha256")},
                       "Compiler source path is not the exact selected source")
        witnesses = w3_compiler_invocations(log, module, path)
        retained = evidence / (phase + "-compiler-" + str(len(inventory)) + ".SwiftFileList")
        with retained.open("xb") as stream:
            stream.write(path.read_bytes())
        w3_require(w3_sha(retained) == w3_sha(path), "Original compiler input retention differs")
        inventory.append({"path": str(retained), "originalPath": str(path),
                          "sha256": w3_sha(path), "requiredSourcePaths": expected_paths,
                          "compilerInvocation": witnesses[0], "compilerInvocations": witnesses})
    w3_require(inventory, "Actual selected target compiler FileList is absent or incomplete")
    return inventory

def wr_require(condition, message):
    if not condition:
        raise ValueError(message)

def wr_sha(path):
    with pathlib.Path(path).open('rb') as stream:
        return hashlib.file_digest(stream, 'sha256').hexdigest()

def wr_packages(source, derived):
    source, derived = pathlib.Path(source), pathlib.Path(derived)
    lock = source / 'Anytype.xcodeproj/project.xcworkspace/xcshareddata/swiftpm/Package.resolved'
    workspace = derived / 'SourcePackages/workspace-state.json'
    wr_require(workspace.is_file() and not workspace.is_symlink(), 'Original warm SwiftPM workspace is missing')
    pins = json.loads(lock.read_text())['pins']
    expected = {row['identity']: row['state'] for row in pins}
    wr_require(len(expected) == len(pins), 'Duplicate package lock identities')
    observed = json.loads(workspace.read_text())
    rows, local = {}, {}
    for dependency in observed['object']['dependencies']:
        name = dependency['packageRef']['identity']
        state = dependency['state']
        if state['name'] == 'fileSystem':
            path = pathlib.Path(state['path'])
            wr_require(path.is_relative_to(source / 'Dependencies/Hanlin/Packages') and path.is_dir()
                       and not path.is_symlink(), 'Unreviewed local package path')
            local[name] = str(path)
            continue
        wr_require(state['name'] == 'sourceControlCheckout' and name in expected and name not in rows,
                   'Unexpected or duplicate warm package checkout')
        wr_require(state['checkoutState'] == expected[name], 'Workspace state differs from exact package lock: ' + name)
        subpath = pathlib.PurePosixPath(dependency['subpath'])
        wr_require(not subpath.is_absolute() and '..' not in subpath.parts, 'Invalid package checkout path')
        checkout = derived / 'SourcePackages/checkouts' / subpath
        wr_require(checkout.is_dir() and not checkout.is_symlink(), 'Physical checkout missing: ' + name)
        git = ['/usr/bin/git', '--no-replace-objects', '-C', str(checkout)]
        head = subprocess.run(git + ['rev-parse', 'HEAD'], capture_output=True, text=True, timeout=30,
                              env=dict(os.environ, GIT_OPTIONAL_LOCKS='0'))
        status = subprocess.run(git + ['status', '--porcelain', '--untracked-files=all'], capture_output=True, text=True,
                                timeout=30, env=dict(os.environ, GIT_OPTIONAL_LOCKS='0'))
        wr_require(head.returncode == status.returncode == 0 and head.stdout.strip() == expected[name]['revision']
                   and status.stdout == '', 'Actual pinned package checkout differs: ' + name)
        rows[name] = {'path': str(checkout), 'revision': head.stdout.strip(), 'trackedStatus': '',
                      'state': state['checkoutState']}
    wr_require(set(rows) == set(expected), 'Warm checkout set differs from complete package lock')
    wr_require(set(local) == {'hanlinagentsdk', 'hanlinminiappsbridge', 'hanlinplatform', 'hanlinpresentationsdk'},
               'Canonical four-package SDK set differs')
    return {'packageResolvedSHA256': wr_sha(lock), 'workspaceStatePath': str(workspace),
            'workspaceStateSHA256': wr_sha(workspace), 'checkouts': rows, 'localSDKPackages': local}

def wr_no_shared_writer(source, derived):
    source, derived = pathlib.Path(source), pathlib.Path(derived)
    processes = subprocess.check_output(['/bin/ps', '-axo', 'pid=,ppid=,args='], text=True)
    for line in processes.splitlines():
        parts = line.strip().split(None, 2)
        if len(parts) != 3 or int(parts[0]) == os.getpid():
            continue
        cmd = parts[2]
        compiler = any(item in cmd for item in ('xcodebuild', 'swift-frontend', '/swiftc', '/clang', 'XCBBuildService'))
        wr_require(not (compiler and (str(source) in cmd or str(derived) in cmd)), 'Shared source/cache compiler is active')
        wr_require('main16-wave2-20261127/control/launch_app16_when_compiler_admitted.py' not in cmd,
                   'Obsolete queued launcher is active')
    database = derived / 'Build/Intermediates.noindex/XCBuildData/build.db'
    wr_require(database.is_file() and not database.is_symlink(), 'Warm build database is absent')
    opened = subprocess.run(['/usr/sbin/lsof', '-nP', str(database)], capture_output=True, text=True, timeout=30)
    wr_require(opened.returncode == 1 and opened.stdout == opened.stderr == '', 'Warm build database has an owner')
